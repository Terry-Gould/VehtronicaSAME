vehtronica_microcan_fd.pid.0=0xE473

vehtronica_microcan_fd.pid.1=0xE472

# Upload
vehtronica_microcan_fd.upload.tool=bossac18
vehtronica_microcan_fd.upload.protocol=sam-ba
vehtronica_microcan_fd.upload.maximum_size=507904
vehtronica_microcan_fd.upload.offset=0x4000
vehtronica_microcan_fd.upload.use_1200bps_touch=true
vehtronica_microcan_fd.upload.wait_for_upload_port=true

# Build
vehtronica_microcan_fd.build.mcu=cortex-m4
vehtronica_microcan_fd.build.f_cpu=120000000L
vehtronica_microcan_fd.build.usb_product="MicroCAN-FD"
vehtronica_microcan_fd.build.usb_manufacturer="Vehtronica"
vehtronica_microcan_fd.build.board=MicroCAN_FD
vehtronica_microcan_fd.build.core=arduino
# including feather build flags to keep compatibility with sketches and libraries built for Adafruit Feather boards.
vehtronica_microcan_fd.build.extra_flags=-D__SAME51J19A__ -DADAFRUIT_FEATHER_M4_EXPRESS -DADAFRUIT_FEATHER_M4_CAN -Dvehtronica_microcan_fd -D__SAMD51__ -D__FPU_PRESENT -DARM_MATH_CM4 -mfloat-abi=hard -mfpu=fpv4-sp-d16 {build.usb_flags}
vehtronica_microcan_fd.build.ldscript=linker_scripts/gcc/flash_with_bootloader.ld
vehtronica_microcan_fd.build.openocdscript=scripts/openocd/daplink_samd51.cfg
vehtronica_microcan_fd.build.variant=MicroCAN-FD
vehtronica_microcan_fd.build.variant_system_lib=
vehtronica_microcan_fd.build.vid=0x04D8
vehtronica_microcan_fd.build.pid=0xE472
vehtronica_microcan_fd.bootloader.tool=openocd
vehtronica_microcan_fd.bootloader.file=microcan_fd/bootloader-MicroCAN_FD_v1_0_0.bin
vehtronica_microcan_fd.compiler.arm.cmsis.ldflags="-L{runtime.tools.CMSIS-5.4.0.path}/CMSIS/Lib/GCC/" "-L{build.variant.path}" -larm_cortexM4lf_math -mfloat-abi=hard -mfpu=fpv4-sp-d16
